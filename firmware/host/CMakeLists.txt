cmake_minimum_required(VERSION 3.20)
project(quota_monitor_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# cJSON v1.7.19 declares cmake_minimum_required < 3.5, which CMake 4 rejects.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "LVGL host configuration" FORCE)
FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v8.3.11 GIT_SHALLOW TRUE)
set(ENABLE_CJSON_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_CJSON_UTILS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(cjson GIT_REPOSITORY https://github.com/DaveGamble/cJSON.git GIT_TAG v1.7.19 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(lvgl cjson)
find_package(SDL2 CONFIG QUIET)
if(NOT TARGET SDL2::SDL2)
    find_program(SDL2_CONFIG sdl2-config REQUIRED)
    execute_process(COMMAND ${SDL2_CONFIG} --cflags OUTPUT_VARIABLE SDL_CFLAGS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND ${SDL2_CONFIG} --libs OUTPUT_VARIABLE SDL_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    separate_arguments(SDL_CFLAGS UNIX_COMMAND "${SDL_CFLAGS}")
    separate_arguments(SDL_LIBS UNIX_COMMAND "${SDL_LIBS}")
    add_library(quota_sdl INTERFACE)
    target_compile_options(quota_sdl INTERFACE ${SDL_CFLAGS})
    target_link_libraries(quota_sdl INTERFACE ${SDL_LIBS})
    add_library(SDL2::SDL2 ALIAS quota_sdl)
endif()
add_library(quota_model ../shared/quota_model.cpp)
target_include_directories(quota_model PUBLIC ../shared ${cjson_SOURCE_DIR})
target_link_libraries(quota_model PUBLIC cjson)
add_library(quota_ui ../shared/quota_ui.cpp)
target_include_directories(quota_ui PUBLIC ../shared)
target_link_libraries(quota_ui PUBLIC quota_model lvgl)
add_executable(quota_simulator simulator.cpp)
target_link_libraries(quota_simulator PRIVATE quota_ui SDL2::SDL2)
add_executable(quota_model_tests model_tests.cpp)
target_link_libraries(quota_model_tests PRIVATE quota_model)
target_compile_definitions(quota_model_tests PRIVATE QUOTA_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../fixtures")
add_executable(quota_provision_tests ../main/provision.cpp)
target_compile_definitions(quota_provision_tests PRIVATE QUOTA_PROVISION_SELF_TEST)
# Assertions remain enabled even with a Release build.
if(MSVC)
    target_compile_options(quota_model_tests PRIVATE /UNDEBUG)
    target_compile_options(quota_provision_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(quota_model_tests PRIVATE -UNDEBUG)
    target_compile_options(quota_provision_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME quota_model COMMAND quota_model_tests)
add_test(NAME quota_provision COMMAND quota_provision_tests)
